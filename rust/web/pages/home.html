Quick start

Three lines to your first item.

No header, no token, no SDK. Pick your language and go.

curl "https://api.openalbion.com/api/v3/weapons?tier=8"
const res = await fetch("https://api.openalbion.com/api/v3/weapons?tier=8");
const { data } = await res.json();
console.log(data.length);
import requests

res = requests.get(
    "https://api.openalbion.com/api/v3/weapons",
    params={"tier": 8},
)
print(len(res.json()["data"]))
